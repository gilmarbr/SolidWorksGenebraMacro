'------------------------------------------------------------------------------
' RODAPE: logo Autom + autor
' O logo (Autom_Logo.bmp) e procurado na pasta da macro (.swp) e depois em
' Imagens do usuario. Sem o arquivo, o rodape mostra so o texto.
'------------------------------------------------------------------------------
Private Sub MontarRodape(y As Single)
    Dim img As MSForms.Image, lb As MSForms.Label, arq As String
    ' 1) Imagem colocada em modo de design (fica gravada dentro do .swp)
    On Error Resume Next
    Set img = Me.Controls("imgLogo")
    On Error GoTo 0
    ' 2) Senao, cria a imagem e carrega o arquivo .bmp
    If img Is Nothing Then
        Set img = Me.Controls.Add("Forms.Image.1", "imgLogo")
        arq = CaminhoLogo()
        If arq <> "" Then
            On Error Resume Next
            Set img.Picture = LoadPicture(arq)
            On Error GoTo 0
        End If
    End If
    img.Left = 6: img.Top = y: img.Width = 106: img.Height = 40
    img.BorderStyle = 0
    img.PictureSizeMode = 3          ' zoom mantendo a proporcao
    Set lb = Me.Controls.Add("Forms.Label.1", "lblAutor")
    lb.Left = 122: lb.Top = y + 5: lb.Width = 400: lb.Height = 32
    lb.Font.Size = 9
    lb.Caption = "Autor: Gilmar Barbosa" & vbCrLf & "Autom - Gerador de Mecanismo Genebra para SOLIDWORKS"
End Sub

Private Function CaminhoLogo() As String
    Dim p As String, k As Long
    On Error Resume Next
    p = swApp.GetCurrentMacroPathName
    k = InStrRev(p, "\")
    If k > 0 Then
        p = Left$(p, k) & "Autom_Logo.bmp"
        If Dir(p) <> "" Then CaminhoLogo = p: Exit Function
    End If
    p = Environ("USERPROFILE") & "\Pictures\Autom_Logo.bmp"
    If Dir(p) <> "" Then CaminhoLogo = p
End Function

' itens separados por "|". padrao: indice (numero) ou texto.
Private Sub AddCombo(pg As Object, id As String, cap As String, itens As String, padrao As Variant, col As Integer, row As Integer, editavel As Boolean, Optional largura As Single = 0)
    Dim cb As MSForms.ComboBox, v As Variant, i As Long
    AddRotulo pg, id, cap, col, row
    Set cb = pg.Controls.Add("Forms.ComboBox.1", "f_" & id)
    cb.Left = XCol(col) + LBLW + 2: cb.Top = YRow(row)
    If largura > 0 Then cb.Width = largura Else cb.Width = CTLW + 20
    cb.Height = 17
    v = Split(itens, "|")
    For i = LBound(v) To UBound(v)
        cb.AddItem v(i)
    Next
    If editavel Then cb.Style = 0 Else cb.Style = 2
    If VarType(padrao) = vbString Then
        ' seleciona pelo indice: .Text em lista suspensa (Style 2) pode nao pegar
        For i = 0 To cb.ListCount - 1
            If LCase(cb.List(i)) = LCase(CStr(padrao)) Then cb.ListIndex = i: Exit For
        Next
        If cb.ListIndex < 0 And editavel Then cb.Text = padrao
    Else
        cb.ListIndex = CLng(padrao)
    End If
    Registrar id, "C", cb
End Sub

Private Sub Registrar(id As String, kind As String, ctl As Object)

    Campos.Add id, ctl
    Tipos.Add id, kind

Public Sub OnCampo(id As String)
    If Iniciando Then Exit Sub
    If id = "modo" Then AtualizarRotuloRef
    If id = "aoVivo" Or id = "rebuild" Or id = "voltas" Or id = "passos" Or id = "atraso" Then Exit Sub

Public Sub OnScroll(id As String)
    Dim msg As String
    If Iniciando Then Exit Sub
    lblTheta.Caption = scrTheta.Value & U("\u00B0")
End Sub

Public Sub OnBotao(id As String)
    Dim msg As String
    Select Case id
        Case "calc"
            Recalcular
            mp.Value = 5
        Case "gerar"
            LerParametros
            txtRes.Text = "Gerando... aguarde."
            mp.Value = 5
            DoEvents
            txtRes.Text = Gerar()
        Case "salvar"
            SalvarPreset
        Case "carregar"
            CarregarPreset
        Case "procurar"
            ProcurarPasta
        Case "fechar"
            G_PararAnim = True
            Unload Me
        Case "posicionar"
            LerParametros
            msg = PosicionarTheta(CDbl(scrTheta.Value), PB("rebuild"))
            If msg <> "" Then MsgBox msg, vbExclamation
        Case "animar"
            LerParametros
            msg = Animar(PN("voltas"), CLng(PN("passos")), PN("atraso"), _
                         (CLng(PN("sentido")) = 1), CDbl(scrTheta.Value), PB("rebuild"))
            If msg <> "" Then MsgBox msg, vbExclamation
        Case "parar"
            G_PararAnim = True
    End Select
End Sub

' PARAMETROS / CALCULO
'------------------------------------------------------------------------------
Private Function ParseNum(s As String) As Double
    ParseNum = Val(Replace(Trim(s), ",", "."))
End Function

Private Sub LerParametros()
    Dim k As Variant, c As Object
    Set Prm = CreateObject("Scripting.Dictionary")
    For Each k In Campos.Keys
        Set c = Campos(k)
        Select Case Tipos(k)
            Case "N"
                Prm(k) = ParseNum(c.Text)
            Case "T"
                Prm(k & "$") = c.Text
            Case "C"
                Prm(k) = c.ListIndex
                Prm(k & "$") = c.Text
            Case "K"
                If c.Value Then Prm(k) = 1 Else Prm(k) = 0
        End Select
    Next
    ' combos numericos: usam o texto do item
    Prm("pinos") = Val(Prm("pinos$"))
    Prm("nFix") = Val(Prm("nFix$"))
End Sub

Private Sub Recalcular()
    Dim msgs As New Collection, ok As Boolean
    On Error GoTo Falha
    LerParametros
    ok = Calcular(msgs)
    txtRes.Text = TextoResultados(ok, msgs)
    Exit Sub
Falha:
    txtRes.Text = U("Valores inv\u00E1lidos: ") & Err.Description
End Sub

Private Sub AtualizarRotuloRef()
    If lblRef Is Nothing Then Exit Sub
    Select Case Campos("modo").ListIndex
        Case 1: lblRef.Caption = "Raio da manivela r [mm]"
        Case 2: lblRef.Caption = U("Di\u00E2metro externo da roda [mm]")
        Case Else: lblRef.Caption = U("Dist\u00E2ncia entre centros C [mm]")
    End Select
End Sub

Private Sub SalvarPreset()
    Dim fnum As Integer, k As Variant, v As String, p As String
    On Error GoTo Falha
    p = Trim(Campos("pasta").Text)
    If p = "" Then MsgBox U("Informe a pasta de sa\u00EDda."), vbExclamation: Exit Sub
    If Dir(p, vbDirectory) = "" Then MkDir p
    fnum = FreeFile
    Open ArqPreset() For Output As #fnum
    For Each k In Campos.Keys
        If Tipos(k) = "K" Then
            If Campos(k).Value Then v = "1" Else v = "0"
        Else
            v = Campos(k).Text
        End If
        Print #fnum, k & "=" & v
    Next
    Close #fnum
    MsgBox U("Predefini\u00E7\u00E3o salva em:") & vbCrLf & ArqPreset(), vbInformation
    Exit Sub
Falha:
    MsgBox U("N\u00E3o foi poss\u00EDvel salvar: ") & Err.Description, vbExclamation
End Sub

Private Sub CarregarPreset()
    Dim fnum As Integer, linha As String, k As String, v As String, p As Long
    On Error GoTo Falha
    If Dir(ArqPreset()) = "" Then MsgBox U("Arquivo n\u00E3o encontrado:") & vbCrLf & ArqPreset(), vbExclamation: Exit Sub
    Iniciando = True
    fnum = FreeFile
    Open ArqPreset() For Input As #fnum
    Do While Not EOF(fnum)
        Line Input #fnum, linha
        p = InStr(linha, "=")
        If p > 0 Then
            k = Left(linha, p - 1): v = Mid(linha, p + 1)
            If Campos.Exists(k) Then
                If Tipos(k) = "K" Then
                    Campos(k).Value = (v = "1")
                ElseIf Tipos(k) = "C" Then
                    SelecionarCombo Campos(k), v
                Else
                    Campos(k).Text = v
                End If
            End If
        End If
    Loop
    Close #fnum
    Iniciando = False
    AtualizarRotuloRef
    Recalcular
    Exit Sub
Falha:
    Iniciando = False
    MsgBox U("N\u00E3o foi poss\u00EDvel carregar: ") & Err.Description, vbExclamation
End Sub

' Lista suspensa (Style 2): escolhe pelo indice; combo editavel aceita texto livre.
Private Sub SelecionarCombo(cb As Object, v As String)
    Dim i As Long
    For i = 0 To cb.ListCount - 1
        If LCase(cb.List(i)) = LCase(v) Then cb.ListIndex = i: Exit Sub
    Next
    If cb.Style = 0 Then cb.Text = v
End Sub

Private Sub ProcurarPasta()
    Dim sh As Object, fo As Object
    On Error Resume Next
    Set sh = CreateObject("Shell.Application")
    Set fo = sh.BrowseForFolder(0, U("Pasta de sa\u00EDda"), 0)
    If Not fo Is Nothing Then Campos("pasta").Text = fo.Self.Path
End Sub
